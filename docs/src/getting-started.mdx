import { LocalizedMarkdown } from "./documentation/LocalizedMarkdown";
import { Meta } from "@storybook/addon-docs/blocks";

<Meta title="Getting Started" />

<LocalizedMarkdown text={"# Getting Started"} />

<LocalizedMarkdown
  text={
    "This guide uses compiled components and CSS from the installed package. React 19 or later is required. No StyleX build plugin is needed for basic usage."
  }
/>

<LocalizedMarkdown text={"## 1. Install the package"} />

```bash
pnpm add @bolerish/ui
```

<LocalizedMarkdown text={"## 2. Load the styles"} />

<LocalizedMarkdown
  text={
    "Import the stylesheet once in your app’s CSS entry, before app overrides. It includes the reset, component styles, tokens and both themes."
  }
/>

```css
@import "@bolerish/ui/global.css";
```

<LocalizedMarkdown text={"## 3. Use a component"} />

<LocalizedMarkdown
  text={
    "Import components directly from the package. No local component files or import aliases need to be generated."
  }
/>

```tsx
import { Button } from "@bolerish/ui/button";

export function SaveButton() {
  return <Button>Save</Button>;
}
```

<LocalizedMarkdown text={"## 4. Choose a theme"} />

<LocalizedMarkdown
  text={
    "Light is the default. Apply themeClassNames.light or themeClassNames.dark to html so portaled controls inherit the same theme. Your app owns system preference detection, saved preferences and the initial theme. Preserve unrelated html classes when switching, and set color-scheme to match."
  }
/>

```tsx
import { themeClassNames } from "@bolerish/ui/theme";

<html className={themeClassNames.dark}>
  <body>{children}</body>
</html>;
```

<LocalizedMarkdown text={"## Optional editable source"} />

<LocalizedMarkdown
  text={
    "Use the CLI when you need to own and edit the component source. This path requires StyleX integration and uses .bolerish.json. It remains compatible with existing copied components. Package updates do not overwrite local edits."
  }
/>

```bash
pnpm dlx @bolerish/ui init --vite
pnpm dlx @bolerish/ui add button
```

<LocalizedMarkdown
  text={
    "If you author custom xstyle values with stylex.create(), configure StyleX for your build tool. Prebuilt components and theme classes work without it."
  }
/>
